Variables & Types

# Déclaration
$nom    = "NexaCloud"
$port   = 5001
$actif  = $true

# Typage explicite (recommandé en script)
[string]$nom  = "NexaCloud"
[int]$port    = 5001
[bool]$debug  = $false

# Tableau
$niveaux = @("INFO", "WARNING", "ERROR")
$niveaux[0]       # → "INFO"
$niveaux.Count    # → 3

# Hashtable (dictionnaire)
$config = @{
    projet = "NexaCloud"
    port   = 5001
}
$config["port"]   # → 5001
$config.port     # → 5001

Le pipeline orienté objet — la différence clé avec Bash

# PowerShell passe des OBJETS (pas du texte brut) d'une cmdlet à l'autre
Get-ChildItem | Where-Object { $_.Length -gt 1024 } | Sort-Object Length -Descending

# $_ = l'objet courant dans le pipeline
Get-ChildItem "*.log" | ForEach-Object { Write-Output $_.Name }

# Compter des éléments : encapsuler dans () puis .Count
(Get-Content "server.log").Count          # ✅ nombre de lignes
(Select-String "ERROR" "server.log").Count  # ✅ nombre de lignes ERROR

# Select-String retourne des objets MatchInfo, pas du texte !
Select-String "ERROR" "server.log" | ForEach-Object { $_.Line }       # texte
Select-String "ERROR" "server.log" | ForEach-Object { $_.LineNumber }  # n° ligne

# Exporter en CSV propre
Get-ChildItem | Select-Object Name, Length | Export-Csv "data.csv" -NoTypeInformation

Conditions

# if / elseif / else
if ($nb -gt 10) {
    Write-Output "Grand"
} elseif ($nb -eq 5) {
    Write-Output "Exactement 5"
} else {
    Write-Output "Petit"
}

# switch (équivalent d'un if/elseif chaîné)
switch ($niveau) {
    "ERROR"    { Write-Output "Erreur !" }
    "CRITICAL" { Write-Output "CRITIQUE !" }
    default    { Write-Output "Inconnu" }
}

# Tester l'existence d'un fichier
if (Test-Path "server.log") { Write-Output "Existe" }
if (-not (Test-Path "server.log")) { Write-Error "Introuvable"; exit 1 }

Boucles

# foreach (boucle classique)
foreach ($n in @("INFO","ERROR")) {
    Write-Output $n
}

# ForEach-Object (pipeline)
1..5 | ForEach-Object { Write-Output "Tour $_" }
# 1..5 génère @(1,2,3,4,5)

# for (numérique)
for ($i = 1; $i -le 5; $i++) {
    Write-Output "Tour $i"
}

# while
$c = 0
while ($c -lt 3) {
    Write-Output $c
    $c++   # raccourci pour $c = $c + 1
}

Fonctions

# Déclarer une fonction
function Compter-Niveau {
    param([string]$Niveau, [string]$Fichier)
    return (Select-String $Niveau $Fichier).Count
}

# Appel avec paramètres nommés
$nb = Compter-Niveau -Niveau "ERROR" -Fichier "server.log"

# Fonction avec couleur (Write-Host)
function Write-Ok { param($m) Write-Host "[OK] $m" -ForegroundColor Green }

# Paramètre avec valeur par défaut
param([string]$LogFile = "server.log", [int]$Seuil = 3)
# Appel : .\script.ps1 -LogFile "autre.log" -Seuil 5

Syntaxes clés à retenir

# Here-string (texte multi-lignes)
$texte = @"
Projet : $nomProjet
Date   : $(Get-Date -Format 'dd/MM/yyyy')
"@   # ← le "@ doit être EN DÉBUT DE LIGNE

# Sous-expression dans une chaîne
"Lignes : $($contenu.Count)"   # ✅
"Lignes : $contenu.Count"      # ❌ affiche .Count comme texte

# Répétition de chaîne
"=" * 40   # → "======...=" (40 fois)

# Objet personnalisé
[PSCustomObject]@{ Niveau = "ERROR"; Count = 5 }

# Arrondi mathématique
[math]::Round($nb * 100 / $total, 1)

# Découper une chaîne
"PORT=5001" -split "=", 2   # → @("PORT","5001")

# Supprimer l'affichage d'une commande
New-Item "fichier" | Out-Null

Lire, écrire, exporter

# Lire un fichier
$lines = Get-Content "server.log"

# Premières / dernières lignes
Get-Content "server.log" | Select-Object -First 5
Get-Content "server.log" | Select-Object -Last 3

# Écrire dans un fichier (écrase)
$rapport | Out-File "rapport.txt" -Encoding UTF8

# Ajouter à un fichier existant
Add-Content -Path "rapport.txt" -Value "nouvelle ligne"

# CSV propre (sans ligne de type PowerShell)
Get-ChildItem | Select-Object Name, Length | Export-Csv "data.csv" -NoTypeInformation

Lancer un script & politique d'exécution

# Lancer un script (session courante)
.\mon-script.ps1

# Lancer dans une session isolée
pwsh -File mon-script.ps1

# Autoriser les scripts sur Windows (1 seule fois)
Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser
💡 Pas de shebang en PowerShell. L'extension .ps1 suffit.
Sur macOS/Linux, pas de restriction d'exécution.

Pièges courants — erreurs à ne pas faire

Erreur fréquente
# ❌ .Count sur la commande brute
Get-Content "server.log".Count

# ✅ Encapsuler dans ()
(Get-Content "server.log").Count
Erreur fréquente
# ❌ Propriété sans $()
"Total : $contenu.Count"

# ✅ Sous-expression obligatoire
"Total : $($contenu.Count)"
Erreur fréquente
# ❌ "@ pas en début de ligne
$t = @"
  texte
  "@   # ← espaces avant : ERREUR

# ✅ Collé à gauche
$t = @"
  texte
"@
